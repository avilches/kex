# Mover tabs entre workspaces Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Poder llevar una tab a otro workspace arrastrandola (el hover de 600 ms sobre un workspace de la WorkspaceBar lo muestra) o desde el submenu "Move to Workspace" del menu contextual de la tab, sin matar su PTY.

**Architecture:** Una funcion pura `applyMoveTabToWorkspace` (en `useWorkspaces.ts`, apoyada en `lib/tabTransfer.ts`) quita la tab del origen con `applyCloseTab` y la coloca en el destino en un unico `setWorkspaces`, conservando el id de la tab. El arrastre se resuelve con una funcion pura `resolveTabDrop` que ahora ve todos los workspaces; el cambio de workspace por hover vive en la `WorkspaceBar` (que pasa a estar dentro del `WorkspaceDndProvider`). El menu usa un contexto pequeno `WorkspaceMoveContext`.

**Tech Stack:** React 19, TypeScript, `@dnd-kit/core`, zustand, vitest, Tailwind v4, shadcn `ContextMenu`.

```ts
import { allPanes, findPane, splitPaneAndInsertTab, updatePane } from "./splitNode";
import type { Tab, Workspace } from "./types";

export type SplitDirection = "left" | "right" | "top" | "bottom";

export type TabPlacement =
  | { kind: "pane"; paneId: string; index?: number }
  | { kind: "split"; paneId: string; direction: SplitDirection; splitId: string; newPaneId: string }
  | { kind: "auto"; splitId: string; newPaneId: string };

export type TabMoveRequest =
  | { kind: "pane"; workspaceId: string; paneId: string; index?: number }
  | { kind: "split"; workspaceId: string; paneId: string; direction: SplitDirection }
  | { kind: "auto"; workspaceId: string }
  | { kind: "new" };

export function workspaceTitleFromCwd(cwd?: string): string {
  return cwd ? (cwd.split(/[\\/]/).filter(Boolean).slice(-1)[0] ?? "shell") : "shell";
}

function insertIntoPane(ws: Workspace, paneId: string, tab: Tab, index?: number): Workspace | null {
  if (!findPane(ws.paneTree, paneId)) return null;
  const paneTree = updatePane(ws.paneTree, paneId, (p) => {
    const at = index === undefined ? p.tabs.length : Math.min(index, p.tabs.length);
    const tabs = [...p.tabs];
    tabs.splice(at, 0, tab);
    return { ...p, tabs, activeTabId: tab.id };
  });
  return { ...ws, paneTree, activePaneId: paneId };
}

function splitInto(
  ws: Workspace,
  paneId: string,
  direction: SplitDirection,
  splitId: string,
  newPaneId: string,
  tab: Tab,
  paneLimit: number,
): Workspace | null {
  if (!findPane(ws.paneTree, paneId)) return null;
  if (allPanes(ws.paneTree).length >= paneLimit) return null;
  const orientation = direction === "left" || direction === "right" ? "horizontal" : "vertical";
  const position = direction === "left" || direction === "top" ? "first" : "second";
  const paneTree = splitPaneAndInsertTab(ws.paneTree, paneId, splitId, newPaneId, orientation, position, tab);
  return { ...ws, paneTree, activePaneId: newPaneId };
}

export function attachTab(ws: Workspace, tab: Tab, placement: TabPlacement, paneLimit: number): Workspace | null {
  switch (placement.kind) {
    case "pane":
      return insertIntoPane(ws, placement.paneId, tab, placement.index);
    case "split":
      return splitInto(ws, placement.paneId, placement.direction, placement.splitId, placement.newPaneId, tab, paneLimit);
    case "auto": {
      const panes = allPanes(ws.paneTree);
      if (panes.length === 1) {
        const only = panes[0]!;
        if (only.tabs.length === 0) return insertIntoPane(ws, only.id, tab);
        return (
          splitInto(ws, only.id, "right", placement.splitId, placement.newPaneId, tab, paneLimit) ??
          insertIntoPane(ws, only.id, tab)
        );
      }
      const target = findPane(ws.paneTree, ws.activePaneId) ?? panes[0]!;
      return insertIntoPane(ws, target.id, tab);
    }
  }
}

export function workspaceForTab(
  tab: Tab,
  ids: { workspaceId: string; paneId: string },
  color: string | null | undefined,
): Workspace {
  const cwd = tab.kind === "terminal" ? tab.cwd : undefined;
  return {
    id: ids.workspaceId,
    title: workspaceTitleFromCwd(cwd),
    cwd,
    color,
    paneTree: { kind: "pane", id: ids.paneId, tabs: [tab], activeTabId: tab.id },
    activePaneId: ids.paneId,
  };
}
```

```ts
export type TabMoveDestination =
  | { kind: "existing"; workspaceId: string; placement: TabPlacement }
  | { kind: "new"; workspaceId: string; paneId: string; color: string | null | undefined };

// Moves a tab to another workspace in one pass so the tab keeps its id (and so
// its PTY). Returns the same array when the move is invalid, so the tab is
// never dropped from the source without having landed somewhere.
export function applyMoveTabToWorkspace(
  workspaces: Workspace[],
  sourceWorkspaceId: string,
  tabId: string,
  dest: TabMoveDestination,
  paneLimit: number,
): Workspace[] {
  const source = workspaces.find((w) => w.id === sourceWorkspaceId);
  const found = source ? findTabPane(source.paneTree, tabId) : null;
  if (!found) return workspaces;
  const { tab } = found;

  if (dest.kind === "new") {
    const detached = applyCloseTab(workspaces, sourceWorkspaceId, tabId);
    return insertWorkspaceAfterActive(detached, sourceWorkspaceId, workspaceForTab(tab, dest, dest.color));
  }

  if (dest.workspaceId === sourceWorkspaceId) return workspaces;
  const target = workspaces.find((w) => w.id === dest.workspaceId);
  const attached = target ? attachTab(target, tab, dest.placement, paneLimit) : null;
  if (!attached) return workspaces;
  return applyCloseTab(workspaces, sourceWorkspaceId, tabId).map((w) =>
    w.id === dest.workspaceId ? attached : w,
  );
}
```

```tsx
type WorkspaceMoveValue = {
  targets: WorkspaceMoveTarget[];
  moveTab: (sourceWorkspaceId: string, tabId: string, request: TabMoveRequest) => void;

  targets: [],
  moveTab: () => {},
});

export function useWorkspaceMove(): WorkspaceMoveValue {
  return useContext(WorkspaceMoveContext);
}

type Props = {
  workspaces: Workspace[];
  onMoveTab: (sourceWorkspaceId: string, tabId: string, request: TabMoveRequest) => void;
  children: ReactNode;

// The targets list only changes when a workspace is added, removed, renamed or
// recolored, so it is memoized on a key instead of on the workspaces array
// (which changes on every cwd update and would re-render every tab).
export function WorkspaceMoveProvider({ workspaces, onMoveTab, children }: Props) {
  const key = workspaces
    .map((w) => `${w.id}\u0000${w.title}\u0000${resolveWorkspaceColor(w.color, w.id) ?? ""}`)
    .join("\u0001");

        id: w.id,
        title: w.title || "shell",
        color: resolveWorkspaceColor(w.color, w.id),
      })),
    [key],
  );
```

```markdown
## Moving a tab to another workspace

`applyMoveTabToWorkspace` (`useWorkspaces.ts`, pure) removes the tab from the source workspace with
`applyCloseTab` and places it in the destination in a single `setWorkspaces`. The tab keeps its id, so a

"New", then every other workspace with a dot of its color. Choosing a workspace uses the `auto`

user stays in the current workspace. "New" creates a workspace right after the source (same group) whose
only tab is the moved one, titled and rooted after the tab's cwd when it is a terminal. A source
workspace that loses its last tab is not closed: it keeps an empty pane. `App.tsx` also calls
`useAgentStore.reassignTabWorkspace` so the agent session and its notifications point at the new
workspace.
```

```ts
export const HOVER_SWITCH_DELAY_MS = 600;

export type HoverSwitch = {
  enter: (id: string) => void;
  leave: (id: string) => void;
  cancel: () => void;
};

export function createHoverSwitch(
  onSwitch: (id: string) => void,
  delayMs: number = HOVER_SWITCH_DELAY_MS,
): HoverSwitch {
  let pending: { id: string; timer: ReturnType<typeof setTimeout> } | null = null;

  const cancel = () => {
    if (!pending) return;
    clearTimeout(pending.timer);
    pending = null;
  };

  return {
    enter(id) {
      if (pending?.id === id) return;
      cancel();
      pending = {
        id,
        timer: setTimeout(() => {
          pending = null;
          onSwitch(id);
        }, delayMs),
      };
    },
    leave(id) {
      if (pending?.id === id) cancel();
    },
    cancel,
  };
}
```

```ts
import { allPanes, findPane } from "./splitNode";
import type { SplitDirection, TabMoveRequest } from "./tabTransfer";
import type { Workspace } from "./types";

export type TabLocation = { paneId: string; wsId: string };

export type TabDropAction =
  | { kind: "reorder"; workspaceId: string; tabId: string; index: number }
  | { kind: "move"; workspaceId: string; tabId: string; paneId: string; index?: number }
  | { kind: "split"; workspaceId: string; tabId: string; paneId: string; direction: SplitDirection }
  | { kind: "transfer"; sourceWorkspaceId: string; tabId: string; request: TabMoveRequest };

const DIRECTIONS: ReadonlySet<string> = new Set(["top", "bottom", "left", "right"]);

export function resolveTabDrop(
  overId: string,
  tabId: string,
  workspaces: Workspace[],
  idx: Map<string, TabLocation>,
  paneLimit: number,
): TabDropAction | null {
  const source = idx.get(tabId);
  if (!source) return null;

  if (overId.startsWith("tab-insert:")) {
    const [, refTabId, side] = overId.split(":");
    if (!refTabId || (side !== "before" && side !== "after")) return null;
    const target = idx.get(refTabId);
    if (!target) return null;
    const targetWs = workspaces.find((w) => w.id === target.wsId);
    const pane = targetWs ? findPane(targetWs.paneTree, target.paneId) : null;
    const refIndex = pane ? pane.tabs.findIndex((t) => t.id === refTabId) : -1;
    if (refIndex === -1) return null;
    const index = refIndex + (side === "after" ? 1 : 0);
    if (target.wsId !== source.wsId) {
      return {
        kind: "transfer",
        sourceWorkspaceId: source.wsId,
        tabId,
        request: { kind: "pane", workspaceId: target.wsId, paneId: target.paneId, index },
      };
    }
    if (source.paneId === target.paneId) return { kind: "reorder", workspaceId: source.wsId, tabId, index };
    return { kind: "move", workspaceId: source.wsId, tabId, paneId: target.paneId, index };
  }

  if (!overId.startsWith("zone:")) return null;
  const [, paneId, zone] = overId.split(":");
  if (!paneId || !zone) return null;
  const targetWs = workspaces.find((w) => findPane(w.paneTree, paneId));
  if (!targetWs) return null;
  const crossing = targetWs.id !== source.wsId;

  if (zone === "center") {
    if (crossing) {
      return {
        kind: "transfer",
        sourceWorkspaceId: source.wsId,
        tabId,
        request: { kind: "pane", workspaceId: targetWs.id, paneId },
      };
    }
    if (source.paneId === paneId) return null;
    return { kind: "move", workspaceId: source.wsId, tabId, paneId, index: undefined };
  }

  if (!DIRECTIONS.has(zone)) return null;
  if (allPanes(targetWs.paneTree).length >= paneLimit) return null;
  const direction = zone as SplitDirection;
  if (crossing) {
    return {
      kind: "transfer",
      sourceWorkspaceId: source.wsId,
      tabId,
      request: { kind: "split", workspaceId: targetWs.id, paneId, direction },
    };
  }
  return { kind: "split", workspaceId: source.wsId, tabId, paneId, direction };
}

export function originToRestore(originId: string | null, currentId: string, placed: boolean): string | null {
  if (placed || !originId || originId === currentId) return null;
  return originId;
}
```
