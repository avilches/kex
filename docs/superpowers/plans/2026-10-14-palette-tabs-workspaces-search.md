# Palette tab and workspace search Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The command palette filters tabs, workspaces, actions and settings in one box, shows the 8 most recently used tabs when empty, and previews a tab's note on selection.

**Architecture:** Pure functions build and rank the tab/workspace rows (`tabEntries.ts`, `tabRank.ts`, `selectEntry.ts`); a small `localStorage` map records when each tab was last used (`workspaces/lib/tabUse.ts`); the action catalog is generated from `SHORTCUTS` plus `shortcutHandlers` already living in `App.tsx`, so no per-shortcut context fields are added. `CommandPalette.tsx` and `App.tsx` stay thin wiring.

**Tech Stack:** TypeScript, React 19, cmdk (`@/components/ui/command`), vitest (`happy-dom` per file where a DOM is needed), hugeicons, pnpm.

**Spec:** `docs/superpowers/specs/2026-10-14-palette-tabs-workspaces-search-design.md` (task: TASK-833). Task 9 amends the spec where this plan deliberately differs.

## Global Constraints

- No em-dash and no emojis anywhere (code, comments, commits, docs).
- Imports use `@/...`, never relative across modules. Relative `./` is fine inside one module folder.
- Commit messages are English, one line, no `Co-Authored-By`, no "Generated by".
- Comments only for a non-obvious *why*, one or two lines.
- pnpm only. Checks before done: `pnpm lint`, `pnpm check-types`, `pnpm test`.
- No backward-compat code, no migrations.
- Entity ids come from `src/lib/ids.ts`; this plan creates no new entities.
- The 8-tab limit applies only to the empty box. While typing, the list is not limited.
- Modes `#` (content), `>` (history), `?` and the themes subpage behave exactly as today.
- Work stays in the worktree `.claude/worktrees/palette-tabs-search`. At the end merge directly into `main`, no PR.

## Review Focus

- A query with regex or glob characters (`c++`, `a.b`, `[x`, `(`) must not throw and must still match literally.
- A corrupt or inaccessible `localStorage` value (`kex-tab-use`) must read as empty, never crash the palette.
- A tab with an empty title and no cwd (blank browser tab) must still render a readable row.
- Choosing a tab whose workspace is hibernated must wake first and navigate second; if the wake rejects, no navigation happens.
- Two tabs with the same title in the same workspace must be distinguishable in the list.
- 1000 tabs must rank in well under one keystroke budget (20 ms).

```ts
// Non-critical, shared-by-all-windows recency of each tab. localStorage keeps it out of
// the workspace JSON: switching tabs must not make the persisted body dirty.

const KEY = "kex-tab-use";
export const TAB_USE_MAX_ENTRIES = 300;

export type TabUseMap = Record<string, number>;

function read(): TabUseMap {
  try {
    const raw = localStorage.getItem(KEY);
    if (!raw) return {};
    const parsed: unknown = JSON.parse(raw);
    if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) return {};
    return parsed as TabUseMap;
  } catch {
    return {};
  }
}

function write(map: TabUseMap): void {
  try {
    localStorage.setItem(KEY, JSON.stringify(map));
  } catch {
    /* ignore */
  }
}

export function recordTabUse(tabId: string, now = Date.now()): void {
  const map = read();
  map[tabId] = now;
  const ids = Object.keys(map);
  if (ids.length > TAB_USE_MAX_ENTRIES) {
    ids.sort((a, b) => map[a] - map[b]);
    for (const id of ids.slice(0, ids.length - TAB_USE_MAX_ENTRIES)) delete map[id];
  }
  write(map);
}

export function tabUseSnapshot(): TabUseMap {
  return read();
}

export function pruneTabUse(liveIds: ReadonlySet<string>): void {
  const map = read();
  let changed = false;
  for (const id of Object.keys(map)) {
    if (!liveIds.has(id)) {
      delete map[id];
      changed = true;
    }
  }
  if (changed) write(map);
}
```

```ts
import type { AgentEntryVisual } from "@/modules/agents/lib/notificationList";
import { allPanes } from "@/modules/workspaces/lib/splitNode";
import { tabDir } from "@/modules/workspaces/lib/titleVars";
import type { Tab, Workspace } from "@/modules/workspaces/lib/types";

export type TabSources = {
  displayTitle(tab: Tab): string;
  git(dir: string | undefined): { branch: string; repoRoot: string };
  runningCommand(tabId: string): string | null;
  agentOf(tabId: string): string | undefined;
  visualOf(tabId: string): AgentEntryVisual | null;
  lastUsed(tabId: string): number;
  workspaceActiveAt(workspaceId: string): number;
};

export type TabEntry = {
  tabId: string;
  workspaceId: string;
  workspaceTitle: string;
  hibernated: boolean;
  kind: Tab["kind"];
  title: string;
  note: string | null;
  cwd: string;
  worktree: string;
  branch: string;
  process: string;
  indicator: AgentEntryVisual | null;
  lastUsedAt: number;
  workspaceActiveAt: number;
  order: number;
};

export type WorkspaceEntry = {
  workspaceId: string;
  title: string;
  color: string | null;
  hibernated: boolean;
  tabCount: number;
  lastUsedAt: number;
};

const WORKTREE_RE = /[\\/]\.claude[\\/]worktrees[\\/]([^\\/]+)/;

export function worktreeName(path: string): string {
  return WORKTREE_RE.exec(path)?.[1] ?? "";
}

function processOf(tab: Tab, src: TabSources): string {
  const asleep = tab.kind === "terminal" ? tab.hibernation : undefined;
  const command = src.runningCommand(tab.id) ?? asleep?.command ?? "";
  const agent = src.agentOf(tab.id) ?? asleep?.agent ?? "";
  return [command.trim(), agent].filter(Boolean).join(" ");
}

export function buildTabEntries(workspaces: readonly Workspace[], src: TabSources): TabEntry[] {
  const out: TabEntry[] = [];
  let order = 0;
  for (const ws of workspaces) {
    const workspaceActiveAt = src.workspaceActiveAt(ws.id);
    for (const pane of allPanes(ws.paneTree)) {
      for (const tab of pane.tabs) {
        const cwd = tabDir(tab) ?? "";
        const git = src.git(cwd || undefined);
        out.push({
          tabId: tab.id,
          workspaceId: ws.id,
          workspaceTitle: ws.title,
          hibernated: !!ws.hibernated,
          kind: tab.kind,
          title: src.displayTitle(tab),
          note: tab.note?.trim() ? tab.note : null,
          cwd,
          worktree: worktreeName(git.repoRoot) || worktreeName(cwd),
          branch: git.branch,
          process: processOf(tab, src),
          indicator: src.visualOf(tab.id),
          lastUsedAt: src.lastUsed(tab.id),
          workspaceActiveAt,
          order: order++,
        });
      }
    }
  }
  return out;
}

export function buildWorkspaceEntries(workspaces: readonly Workspace[], src: TabSources): WorkspaceEntry[] {
  return workspaces.map((ws) => {
    const tabs = allPanes(ws.paneTree).flatMap((p) => p.tabs);
    const newestTab = tabs.reduce((max, t) => Math.max(max, src.lastUsed(t.id)), 0);
    return {
      workspaceId: ws.id,
      title: ws.title,
      color: ws.color ?? null,
      hibernated: !!ws.hibernated,
      tabCount: tabs.length,
      lastUsedAt: newestTab || src.workspaceActiveAt(ws.id),
    };
  });
}

export function duplicateTitleIds(entries: readonly TabEntry[]): Set<string> {
  const groups = new Map<string, string[]>();
  for (const e of entries) {
    const key = `${e.workspaceId}\u0000${e.title}`;
```

```ts
import { fuzzyScoreLower } from "./fuzzy";
import type { TabEntry, WorkspaceEntry } from "./tabEntries";

export const RECENT_LIMIT = 8;

// Best case for one query character in fuzzyScoreLower: 1 + boundary + consecutive.
const MAX_PER_CHAR = 19;

type Tier = 0 | 1 | 2 | 3;
type Hit = { tier: Tier; score: number };

export function byRecency(a: TabEntry, b: TabEntry): number {
  return (
    Number(a.hibernated) - Number(b.hibernated) ||
    b.lastUsedAt - a.lastUsedAt ||
    b.workspaceActiveAt - a.workspaceActiveAt ||
    a.order - b.order
  );
}

export function recentTabs(entries: readonly TabEntry[], limit = RECENT_LIMIT): TabEntry[] {
  return [...entries].sort(byRecency).slice(0, limit);
}

function fuzzy(query: string, field: string): number | null {
  const s = fuzzyScoreLower(query, field.toLowerCase());
  return s === null ? null : Math.min(1, s / (query.length * MAX_PER_CHAR));
}

// Long free-form fields (paths, notes, commands) match by substring: a fuzzy
// subsequence over a long path matches almost any short query.
function substring(query: string, field: string): number | null {
  if (!field) return null;
  const i = field.toLowerCase().indexOf(query);
  return i < 0 ? null : 1 - i / (field.length + 1);
}

function best(...scores: (number | null)[]): number | null {
  let top: number | null = null;
  for (const s of scores) if (s !== null && (top === null || s > top)) top = s;
  return top;
}

function hitOf(e: TabEntry, q: string): Hit | null {
  const primary = best(fuzzy(q, e.title), substring(q, e.note ?? ""));
  if (primary !== null) return { tier: 0, score: primary };
  const workspace = fuzzy(q, e.workspaceTitle);
  if (workspace !== null) return { tier: 1, score: workspace };
  const location = best(substring(q, e.cwd), substring(q, e.worktree), substring(q, e.branch));
  if (location !== null) return { tier: 2, score: location };
  const process = substring(q, e.process);
  if (process !== null) return { tier: 3, score: process };
  return null;
}

export function rankTabEntries(entries: readonly TabEntry[], term: string): TabEntry[] {
  const q = term.trim().toLowerCase();
  if (!q) return recentTabs(entries);
  const hits: { entry: TabEntry; hit: Hit }[] = [];
  for (const entry of entries) {
    const hit = hitOf(entry, q);
    if (hit) hits.push({ entry, hit });
  }
  hits.sort((a, b) => a.hit.tier - b.hit.tier || b.hit.score - a.hit.score || byRecency(a.entry, b.entry));
  return hits.map((h) => h.entry);
}

export function rankWorkspaceEntries(entries: readonly WorkspaceEntry[], term: string): WorkspaceEntry[] {
  const q = term.trim().toLowerCase();
  if (!q) return [];
  const hits: { entry: WorkspaceEntry; score: number }[] = [];
  for (const entry of entries) {
    const score = fuzzy(q, entry.title);
    if (score !== null) hits.push({ entry, score });
  }
  hits.sort(
    (a, b) =>
      b.score - a.score ||
      Number(a.entry.hibernated) - Number(b.entry.hibernated) ||
      b.entry.lastUsedAt - a.entry.lastUsedAt,
  );
  return hits.map((h) => h.entry);
}
```

```ts
import type { TabEntry, WorkspaceEntry } from "./tabEntries";

// A hibernated workspace keeps its tabs in the tree, so navigating after the wake always finds the tab.
export async function selectTab(
  entry: Pick<TabEntry, "tabId" | "workspaceId" | "hibernated">,
  deps: { wake(workspaceId: string): Promise<void>; goToTab(tabId: string): void },
```
